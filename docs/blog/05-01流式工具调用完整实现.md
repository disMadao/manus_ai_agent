## 总结

这里贴一个使用anthropic sdks java版本实现的仿照claude code流式进行工具调用的完整代码，原理是sdk 中做了流式的json传输，规定了几种json内容，可以做到将工具调用作为blcok块传输，通过strat,end之类的标记，blcok完整了就能立马进行工具调用了。

## 流式执行工具agent 

```java
import com.anthropic.client.AnthropicClient;
import com.anthropic.client.okhttp.AnthropicOkHttpClient;
import com.anthropic.models.messages.*;

import java.util.*;
import java.util.concurrent.*;

/**
 * 流式工具执行 Agent
 * 
 * 特点：
 * 1. 模型输出文本时实时显示
 * 2. 模型输出工具调用时，一完成就立即异步执行（不等其他工具）
 * 3. 自动循环：有工具调用 → 执行 → 结果发回 → 继续
 */
public class StreamingAgent {

    private final AnthropicClient client;
    private final List<Tool> tools;
    private final Map<String, ToolExecutor> toolExecutors;
    private final ExecutorService executor = Executors.newCachedThreadPool();

    public StreamingAgent(AnthropicClient client) {
        this.client = client;
        this.tools = new ArrayList<>();
        this.toolExecutors = new HashMap<>();
    }

    // ==================== 工具注册 ====================

    /**
     * 注册工具
     */
    public void registerTool(String name, String description, 
                             Map<String, Object> parameters, 
                             ToolExecutor executor) {
        Tool tool = Tool.builder()
            .name(name)
            .description(description)
            .inputSchema(Tool.InputSchema.builder()
                .type(Tool.InputSchema.Type.OBJECT)
                .properties(parameters)
                .build())
            .build();
        
        tools.add(tool);
        toolExecutors.put(name, executor);
    }

    // ==================== Agent 主循环 ====================

    /**
     * 运行 Agent（自动循环直到没有工具调用）
     */
    public String run(String userMessage) {
        List<MessageParam> messages = new ArrayList<>();
        messages.add(MessageParam.ofUser(userMessage));

        int maxIterations = 10;
        
        for (int i = 0; i < maxIterations; i++) {
            System.out.println("\n========== 迭代 " + (i + 1) + " ==========");
            
            IterationResult result = runOneIteration(messages);
            
            if (result.toolResults.isEmpty()) {
                // 没有工具调用，返回最终文本
                return result.fullText;
            }
            
            // 有工具调用，添加 assistant 消息和 tool_result 消息
            messages.add(MessageParam.ofAssistant(result.assistantContent));
            
            // 构建 tool_result 消息
            List<ToolResultBlockParam> toolResultBlocks = new ArrayList<>();
            for (var entry : result.toolResults.entrySet()) {
                toolResultBlocks.add(ToolResultBlockParam.builder()
                    .toolUseId(entry.getKey())
                    .content(entry.getValue())
                    .build());
            }
            messages.add(MessageParam.ofUser(toolResultBlocks));
        }
        
        return "[达到最大迭代次数]";
    }

    // ==================== 单次迭代（流式处理） ====================

    private IterationResult runOneIteration(List<MessageParam> messages) {
        MessageCreateParams params = MessageCreateParams.builder()
            .model("claude-sonnet-4-20250514")
            .maxTokens(4096)
            .tools(tools)
            .messages(messages)
            .build();

        // 状态追踪
        Map<Integer, BlockBuilder> blocks = new ConcurrentHashMap<>();
        StringBuilder fullText = new StringBuilder();
        List<ContentBlock> assistantContent = Collections.synchronizedList(new ArrayList<>());
        Map<String, CompletableFuture<String>> pendingTools = new ConcurrentHashMap<>();
        CountDownLatch streamDone = new CountDownLatch(1);

        // 流式处理
        client.messages().createStreaming(params).subscribe(
            event -> handleEvent(event, blocks, fullText, assistantContent, pendingTools),
            error -> {
                System.err.println("流式错误: " + error.getMessage());
                streamDone.countDown();
            },
            () -> streamDone.countDown()
        );

        // 等待流结束
        try {
            streamDone.await();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        // 等待所有工具执行完成，收集结果
        Map<String, String> toolResults = new HashMap<>();
        for (var entry : pendingTools.entrySet()) {
            try {
                toolResults.put(entry.getKey(), entry.getValue().get(30, TimeUnit.SECONDS));
            } catch (Exception e) {
                toolResults.put(entry.getKey(), "Error: " + e.getMessage());
            }
        }

        return new IterationResult(fullText.toString(), assistantContent, toolResults);
    }

    // ==================== 事件处理（核心） ====================

    private void handleEvent(RawMessageStreamEvent event,
                            Map<Integer, BlockBuilder> blocks,
                            StringBuilder fullText,
                            List<ContentBlock> assistantContent,
                            Map<String, CompletableFuture<String>> pendingTools) {
        
        // 根据事件类型处理（需要根据实际 SDK 类型调整）
        String eventType = getEventType(event);
        
        switch (eventType) {
            case "content_block_start" -> {
                int index = getIndex(event);
                Object contentBlock = getContentBlock(event);
                String blockType = getBlockType(contentBlock);
                
                BlockBuilder builder = new BlockBuilder();
                builder.type = blockType;
                builder.index = index;
                
                if ("tool_use".equals(blockType)) {
                    builder.toolId = getToolId(contentBlock);
                    builder.toolName = getToolName(contentBlock);
                    builder.inputJson = new StringBuilder();
                    System.out.println("\n[工具调用开始] " + builder.toolName);
                } else if ("text".equals(blockType)) {
                    builder.text = new StringBuilder();
                }
                
                blocks.put(index, builder);
            }
            
            case "content_block_delta" -> {
                int index = getIndex(event);
                BlockBuilder builder = blocks.get(index);
                if (builder == null) return;
                
                Object delta = getDelta(event);
                String deltaType = getDeltaType(delta);
                
                if ("text_delta".equals(deltaType)) {
                    String text = getDeltaText(delta);
                    builder.text.append(text);
                    fullText.append(text);
                    System.out.print(text);  // 实时输出
                } else if ("input_json_delta".equals(deltaType)) {
                    String partialJson = getPartialJson(delta);
                    builder.inputJson.append(partialJson);
                }
            }
            
            case "content_block_stop" -> {
                int index = getIndex(event);
                BlockBuilder builder = blocks.remove(index);
                if (builder == null) return;
                
                if ("tool_use".equals(builder.type)) {
                    // 工具块完整了！
                    String toolName = builder.toolName;
                    String toolId = builder.toolId;
                    String inputJson = builder.inputJson.toString();
                    
                    System.out.println("\n[工具参数完整] " + toolName + ": " + inputJson);
                    System.out.println("[立即异步执行...]");
                    
                    // 记录到 assistant content
                    assistantContent.add(createToolUseBlock(toolId, toolName, inputJson));
                    
                    // 立即异步执行！
                    ToolExecutor toolExecutor = toolExecutors.get(toolName);
                    if (toolExecutor != null) {
                        CompletableFuture<String> future = CompletableFuture.supplyAsync(
                            () -> {
                                try {
                                    String result = toolExecutor.execute(inputJson);
                                    System.out.println("[工具完成] " + toolName + " → " + result);
                                    return result;
                                } catch (Exception e) {
                                    return "Error: " + e.getMessage();
                                }
                            },
                            executor
                        );
                        pendingTools.put(toolId, future);
                    }
                } else if ("text".equals(builder.type)) {
                    assistantContent.add(createTextBlock(builder.text.toString()));
                }
            }
            
            case "message_start", "message_delta", "message_stop", "ping" -> {
                // 忽略或记录
            }
            
            default -> {
                // 未知事件类型，忽略
            }
        }
    }

    // ==================== SDK 适配方法（需根据实际 SDK 调整） ====================
    
    // 这些方法需要根据你实际使用的 SDK 版本调整
    // 因为官方没有文档说明具体类名，这里用占位实现
    
    private String getEventType(RawMessageStreamEvent event) {
        // 实际实现：return event.type() 或 event.getType()
        // 或者用 instanceof 判断
        return event.toString(); // 占位
    }
    
    private int getIndex(RawMessageStreamEvent event) {
        // 实际实现：return event.index()
        return 0; // 占位
    }
    
    private Object getContentBlock(RawMessageStreamEvent event) {
        return null; // 占位
    }
    
    private String getBlockType(Object block) {
        return "text"; // 占位
    }
    
    private String getToolId(Object block) {
        return ""; // 占位
    }
    
    private String getToolName(Object block) {
        return ""; // 占位
    }
    
    private Object getDelta(RawMessageStreamEvent event) {
        return null; // 占位
    }
    
    private String getDeltaType(Object delta) {
        return "text_delta"; // 占位
    }
    
    private String getDeltaText(Object delta) {
        return ""; // 占位
    }
    
    private String getPartialJson(Object delta) {
        return ""; // 占位
    }
    
    private ContentBlock createToolUseBlock(String id, String name, String inputJson) {
        return null; // 占位，实际构建 ToolUseBlock
    }
    
    private ContentBlock createTextBlock(String text) {
        return null; // 占位，实际构建 TextBlock
    }

    // ==================== 辅助类 ====================

    private static class BlockBuilder {
        String type;
        int index;
        // text 类型
        StringBuilder text;
        // tool_use 类型
        String toolId;
        String toolName;
        StringBuilder inputJson;
    }

    private static class IterationResult {
        final String fullText;
        final List<ContentBlock> assistantContent;
        final Map<String, String> toolResults;

        IterationResult(String fullText, List<ContentBlock> assistantContent, 
                       Map<String, String> toolResults) {
            this.fullText = fullText;
            this.assistantContent = assistantContent;
            this.toolResults = toolResults;
        }
    }

    @FunctionalInterface
    public interface ToolExecutor {
        String execute(String inputJson) throws Exception;
    }
    
    public void shutdown() {
        executor.shutdown();
    }
}
```

## 使用示例

```java
public class Main {
    public static void main(String[] args) {
        // 创建客户端
        AnthropicClient client = AnthropicOkHttpClient.fromEnv();
        
        // 创建 Agent
        StreamingAgent agent = new StreamingAgent(client);
        
        // 注册工具
        agent.registerTool(
            "read_file",
            "读取文件内容",
            Map.of(
                "path", Map.of("type", "string", "description", "文件路径")
            ),
            inputJson -> {
                // 解析 JSON，执行工具
                String path = parseJson(inputJson).get("path");
                Thread.sleep(500); // 模拟 IO
                return "文件内容: Hello from " + path;
            }
        );
        
        agent.registerTool(
            "write_file",
            "写入文件",
            Map.of(
                "path", Map.of("type", "string"),
                "content", Map.of("type", "string")
            ),
            inputJson -> {
                Thread.sleep(300);
                return "写入成功";
            }
        );
        
        agent.registerTool(
            "search_web",
            "搜索网页",
            Map.of(
                "query", Map.of("type", "string")
            ),
            inputJson -> {
                Thread.sleep(1000); // 模拟网络请求

        );
        
        // 运行
        String result = agent.run("请读取 /tmp/a.txt 和 /tmp/b.txt，然后搜索一下今天的天气");
        
        System.out.println("\n========== 最终结果 ==========");
        System.out.println(result);
        
        agent.shutdown();
    }
```
